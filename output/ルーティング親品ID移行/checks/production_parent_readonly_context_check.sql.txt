    LEFT JOIN approved a ON a.step_id = s.id
    LEFT JOIN m_product op ON BINARY op.product_code = BINARY s.legacy_parent_code
    LEFT JOIN m_product np ON BINARY np.product_code = BINARY a.hierarchy_parent_code
), approval_context AS (
    SELECT a.*, s.id AS actual_step_id, s.legacy_parent_code,
           s.resolved_parent_id, s.proposed_parent_id,
           COALESCE(s.id IS NOT NULL
             AND s.routing_id = a.routing_id
             AND s.output_product_id = a.output_product_id
             AND BINARY s.hierarchy_path = BINARY a.hierarchy_path
             AND (BINARY s.legacy_parent_code = BINARY a.current_parent_code
                  OR BINARY s.legacy_parent_code = BINARY a.hierarchy_parent_code)
             AND BINARY output.product_code = BINARY a.output_product_code
             AND BINARY routing_product.product_code = BINARY a.routing_product_code
             AND BINARY r.routing_code = BINARY a.routing_code, 0) AS context_ok
    FROM approved a
    LEFT JOIN resolved s ON s.id = a.step_id
    LEFT JOIN m_routing r ON r.id = s.routing_id
    LEFT JOIN m_product output ON output.id = s.output_product_id
    LEFT JOIN m_product routing_product ON routing_product.id = r.product_id
), errors AS (
    SELECT step_id, 'approved_missing_or_context_changed' AS error_code
    FROM approval_context WHERE context_ok = 0
    UNION ALL
    SELECT step_id, 'approved_parent_code_unresolved_or_different'
    FROM approval_context
    WHERE actual_step_id IS NOT NULL
      AND (proposed_parent_id IS NULL OR NOT (resolved_parent_id <=> proposed_parent_id))
    UNION ALL
    SELECT s.id, 'unregistered_parent_not_approved'
    FROM resolved s
    WHERE BINARY s.legacy_parent_code <> BINARY '' AND s.registered_legacy_parent_id IS NULL
      AND (s.approved_step_id IS NULL OR NOT (BINARY s.legacy_parent_code <=> BINARY s.proposed_code)
           AND NOT EXISTS (SELECT 1 FROM approved a WHERE a.step_id = s.id AND BINARY a.current_parent_code = BINARY s.legacy_parent_code))
    UNION ALL
    SELECT s.id, 'nonempty_parent_unresolved'
    FROM resolved s WHERE BINARY s.legacy_parent_code <> BINARY '' AND s.resolved_parent_id IS NULL
    UNION ALL
    SELECT s.id, 'source_bom_parent_or_child_mismatch'
    FROM resolved s
    JOIN m_bom_item bi ON bi.id = s.source_bom_item_id
    LEFT JOIN m_bom b ON b.id = bi.bom_id
    WHERE s.resolved_parent_id IS NOT NULL
      AND (NOT (b.parent_product_id <=> s.resolved_parent_id) OR NOT (bi.child_product_id <=> s.output_product_id))
    UNION ALL
    SELECT s.id, 'actual_hierarchy_parent_mismatch'
    FROM resolved s
    WHERE s.resolved_parent_id IS NOT NULL
      AND NOT (s.output_product_id <=> s.resolved_parent_id)
      AND EXISTS (SELECT 1 FROM m_routing_step ps
                  WHERE ps.routing_id = s.routing_id AND BINARY ps.hierarchy_path = BINARY s.actual_parent_path)
      AND NOT EXISTS (SELECT 1 FROM m_routing_step ps
                      WHERE ps.routing_id = s.routing_id AND BINARY ps.hierarchy_path = BINARY s.actual_parent_path
                        AND ps.output_product_id = s.resolved_parent_id)
    UNION ALL
    SELECT s.id, 'approved_parent_material_missing'
    FROM resolved s
    WHERE s.approved_step_id IS NOT NULL AND s.resolved_parent_id IS NOT NULL
      AND NOT EXISTS (SELECT 1 FROM m_routing_step ps JOIN m_routing_step_material mat ON mat.routing_step_id = ps.id
                      WHERE ps.routing_id = s.routing_id AND BINARY ps.hierarchy_path = BINARY s.actual_parent_path
                        AND ps.output_product_id = s.resolved_parent_id AND mat.component_id = s.output_product_id)
    UNION ALL
    SELECT s.id, 'approved_bom_parent_child_missing'
    FROM resolved s
    WHERE s.approved_step_id IS NOT NULL AND s.resolved_parent_id IS NOT NULL
      AND NOT EXISTS (SELECT 1 FROM m_bom_item bi JOIN m_bom b ON b.id = bi.bom_id
                      WHERE b.parent_product_id = s.resolved_parent_id AND bi.child_product_id = s.output_product_id)
)
SELECT DATABASE() AS actual_database,
       (SELECT COUNT(*) FROM legacy) AS total_steps,
       (SELECT COUNT(*) FROM approved) AS approved_count,
       (SELECT COUNT(*) FROM approval_context WHERE actual_step_id IS NOT NULL) AS approved_present_count,
       (SELECT COUNT(*) FROM resolved WHERE resolved_parent_id IS NOT NULL) AS resolved_parent_count,
       (SELECT COUNT(*) FROM resolved WHERE BINARY legacy_parent_code = BINARY '' AND resolved_parent_id IS NULL) AS originally_optional_parent_count,
       COUNT(*) AS error_count, COUNT(DISTINCT step_id) AS error_step_count,
       COALESCE(SUM(error_code = 'approved_missing_or_context_changed'), 0) AS approved_context_errors,
       COALESCE(SUM(error_code = 'approved_parent_code_unresolved_or_different'), 0) AS approved_parent_resolution_errors,
       COALESCE(SUM(error_code = 'unregistered_parent_not_approved'), 0) AS unapproved_parent_errors,
       COALESCE(SUM(error_code = 'nonempty_parent_unresolved'), 0) AS unresolved_parent_errors,
       COALESCE(SUM(error_code = 'source_bom_parent_or_child_mismatch'), 0) AS source_bom_errors,
       COALESCE(SUM(error_code = 'actual_hierarchy_parent_mismatch'), 0) AS hierarchy_parent_errors,
       COALESCE(SUM(error_code = 'approved_parent_material_missing'), 0) AS approved_material_errors,
       COALESCE(SUM(error_code = 'approved_bom_parent_child_missing'), 0) AS approved_bom_errors
FROM errors;

), legacy AS (
    SELECT s.*, REGEXP_REPLACE(COALESCE(s.remark, ''), '^[[:space:]\\x{001C}-\\x{001F}]+|[[:space:]\\x{001C}-\\x{001F}]+$', '') AS legacy_parent_code,
           CASE WHEN LOCATE('.', s.hierarchy_path) > 0
                THEN LEFT(s.hierarchy_path, CHAR_LENGTH(s.hierarchy_path) - CHAR_LENGTH(SUBSTRING_INDEX(s.hierarchy_path, '.', -1)) - 1)
                ELSE NULL END AS actual_parent_path
    FROM m_routing_step s
), resolved AS (
    SELECT s.*, a.step_id AS approved_step_id, a.hierarchy_parent_code AS proposed_code,
           op.id AS registered_legacy_parent_id, np.id AS proposed_parent_id,
           COALESCE(op.id, CASE WHEN BINARY s.legacy_parent_code <> BINARY ''
                   AND BINARY s.legacy_parent_code = BINARY a.current_parent_code THEN np.id ELSE NULL END) AS resolved_parent_id
    FROM legacy s
    LEFT JOIN approved a ON a.step_id = s.id
    LEFT JOIN m_product op ON BINARY op.product_code = BINARY s.legacy_parent_code
    LEFT JOIN m_product np ON BINARY np.product_code = BINARY a.hierarchy_parent_code
), approval_context AS (
    SELECT a.*, s.id AS actual_step_id, s.legacy_parent_code,
           s.resolved_parent_id, s.proposed_parent_id,
           COALESCE(s.id IS NOT NULL
             AND s.routing_id = a.routing_id
             AND s.output_product_id = a.output_product_id
             AND BINARY s.hierarchy_path = BINARY a.hierarchy_path
             AND (BINARY s.legacy_parent_code = BINARY a.current_parent_code
                  OR BINARY s.legacy_parent_code = BINARY a.hierarchy_parent_code)
             AND BINARY output.product_code = BINARY a.output_product_code
             AND BINARY routing_product.product_code = BINARY a.routing_product_code
             AND BINARY r.routing_code = BINARY a.routing_code, 0) AS context_ok
    FROM approved a
    LEFT JOIN resolved s ON s.id = a.step_id
    LEFT JOIN m_routing r ON r.id = s.routing_id
    LEFT JOIN m_product output ON output.id = s.output_product_id
    LEFT JOIN m_product routing_product ON routing_product.id = r.product_id
), errors AS (
    SELECT step_id, 'approved_missing_or_context_changed' AS error_code
    FROM approval_context WHERE context_ok = 0
    UNION ALL
    SELECT step_id, 'approved_parent_code_unresolved_or_different'
    FROM approval_context
    WHERE actual_step_id IS NOT NULL
      AND (proposed_parent_id IS NULL OR NOT (resolved_parent_id <=> proposed_parent_id))
    UNION ALL
    SELECT s.id, 'unregistered_parent_not_approved'
    FROM resolved s
    WHERE BINARY s.legacy_parent_code <> BINARY '' AND s.registered_legacy_parent_id IS NULL
      AND (s.approved_step_id IS NULL OR NOT (BINARY s.legacy_parent_code <=> BINARY s.proposed_code)
           AND NOT EXISTS (SELECT 1 FROM approved a WHERE a.step_id = s.id AND BINARY a.current_parent_code = BINARY s.legacy_parent_code))
    UNION ALL
    SELECT s.id, 'nonempty_parent_unresolved'
    FROM resolved s WHERE BINARY s.legacy_parent_code <> BINARY '' AND s.resolved_parent_id IS NULL
    UNION ALL
    SELECT s.id, 'source_bom_parent_or_child_mismatch'
    FROM resolved s
    JOIN m_bom_item bi ON bi.id = s.source_bom_item_id
    LEFT JOIN m_bom b ON b.id = bi.bom_id
    WHERE s.resolved_parent_id IS NOT NULL
      AND (NOT (b.parent_product_id <=> s.resolved_parent_id) OR NOT (bi.child_product_id <=> s.output_product_id))
    UNION ALL
    SELECT s.id, 'actual_hierarchy_parent_mismatch'
    FROM resolved s
    WHERE s.resolved_parent_id IS NOT NULL
      AND NOT (s.output_product_id <=> s.resolved_parent_id)
      AND EXISTS (SELECT 1 FROM m_routing_step ps
                  WHERE ps.routing_id = s.routing_id AND BINARY ps.hierarchy_path = BINARY s.actual_parent_path)
      AND NOT EXISTS (SELECT 1 FROM m_routing_step ps
                      WHERE ps.routing_id = s.routing_id AND BINARY ps.hierarchy_path = BINARY s.actual_parent_path
                        AND ps.output_product_id = s.resolved_parent_id)
    UNION ALL
    SELECT s.id, 'approved_parent_material_missing'
    FROM resolved s
    WHERE s.approved_step_id IS NOT NULL AND s.resolved_parent_id IS NOT NULL
      AND NOT EXISTS (SELECT 1 FROM m_routing_step ps JOIN m_routing_step_material mat ON mat.routing_step_id = ps.id
                      WHERE ps.routing_id = s.routing_id AND BINARY ps.hierarchy_path = BINARY s.actual_parent_path
                        AND ps.output_product_id = s.resolved_parent_id AND mat.component_id = s.output_product_id)
    UNION ALL
    SELECT s.id, 'approved_bom_parent_child_missing'
    FROM resolved s
    WHERE s.approved_step_id IS NOT NULL AND s.resolved_parent_id IS NOT NULL
      AND NOT EXISTS (SELECT 1 FROM m_bom_item bi JOIN m_bom b ON b.id = bi.bom_id
                      WHERE b.parent_product_id = s.resolved_parent_id AND bi.child_product_id = s.output_product_id)
)
SELECT a.step_id, a.routing_id, a.routing_code, a.routing_product_code,
       a.output_product_id, a.output_product_code, a.hierarchy_path,
       a.current_parent_code AS approved_old_code, a.legacy_parent_code AS current_old_code,
       a.hierarchy_parent_code AS proposed_parent_code, a.proposed_parent_id AS production_new_parent_id,
       a.resolved_parent_id AS command_resolved_parent_id, a.context_ok,
       (SELECT COUNT(*) FROM errors e WHERE e.step_id = a.step_id) AS error_count
FROM approval_context a ORDER BY a.step_id;
