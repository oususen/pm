284def _get_max_parent_bom_lead_time(product_id):
285 """
286 製品の完成品向け累積LT（自分LTを含む）を取得する。
287
288 RoutingStep（output_product=対象製品）から、完成品までの階層累積LTの最大値を使用。
289 final工程のLTを起点に、hierarchy_path で子階層へ伝播した total LT を採用。
290 対象製品の候補工程が複数ある場合は最大値。
291
292 Args:
293 product_id: 製品ID
294
295 Returns:
296 int: 最大リードタイム（日）
297 """
298 from masters.models import RoutingStep
299 from masters.services.routing_service import build_effective_routing_q
300
301 if not product_id:
302 return 0
303
304 step_qs = RoutingStep.objects.filter(
305 output_product_id=product_id,
306 ).filter(
307 build_effective_routing_q(prefix='routing__')
308 ).select_related('line', 'output_product')
309
310 candidate_steps = list(step_qs)
311 if not candidate_steps:
312 return 0
313
314 routing_ids = {s.routing_id for s in candidate_steps if s.routing_id}
315 if not routing_ids:
316 return 0
317
318 steps = list(
319 RoutingStep.objects.filter(routing_id__in=routing_ids)
320 .filter(build_effective_routing_q(prefix='routing__'))
321 .select_related('line', 'output_product')
322 )

324 def path_key(path):
325 try:
326 return tuple(int(p) for p in str(path).split('.'))
327 except Exception:
328 return (str(path),)
329
330 lt_by_step = {}
331 for routing_id in routing_ids:
332 routing_steps = [s for s in steps if s.routing_id == routing_id]
333 if not routing_steps:
334 continue
335
336 step_map = {
337 s.hierarchy_path: s
338 for s in routing_steps
339 if s.hierarchy_path and s.hierarchy_path != 'final'
340 }
341 children_map = {}
342 for path in step_map.keys():
343 parent_path = path.rsplit('.', 1)[0] if '.' in path else None
344 children_map.setdefault(parent_path, []).append(path)
345
346 final_step = next((s for s in routing_steps if s.hierarchy_path == 'final'), None)
347 base_days = 0
348 if final_step:
349 base_days = resolve_lead_days_for_step(final_step)
350 lt_by_step[final_step.id] = {'total': base_days, 'self': base_days}
351
352 def compute(path, parent_days):
353 step = step_map.get(path)
354 if not step:
355 return
356 self_days = resolve_lead_days_for_step(step)
357 total_days = parent_days + self_days
358 lt_by_step[step.id] = {'total': total_days, 'self': self_days}
359 for child_path in sorted(children_map.get(path, []), key=path_key):
360 compute(child_path, total_days)
361
362 for root_path in sorted(children_map.get(None, []), key=path_key):
363 compute(root_path, base_days)
364
365 # hierarchy_path が無いデータでも最低限 self LT を返せるようにする
366 for step in routing_steps:
367 if step.id in lt_by_step:
368 continue
369 self_days = resolve_lead_days_for_step(step)
370 lt_by_step[step.id] = {'total': self_days, 'self': self_days}
371
372 cumulative_lt = 0
373 for step in candidate_steps:
374 val = (lt_by_step.get(step.id) or {}).get('total')
375 if val is not None and int(val) > cumulative_lt:
376 cumulative_lt = int(val)
377
378 return cumulative_lt

739def _floor_shipping_plan_shipment(backlog, shift_fn, parent_product, qty_per):
740 """フロア配送ラインの出庫をsequence_noベースのLTで計算（pickup側と同一ロジック）
741
742 pickup (views.py) の便判定:
743 sorted by sequence_no → lot_index で判定
744 - sequence_no > 50: 強制2便 (LT=1)
745 - lot_index == 0 かつ sequence_no ≤ 50: 1便 (LT=2)
746 - lot_index > 0 かつ sequence_no ≤ 50: 2便 (LT=1)
747 """
748 total = Decimal('0')
749 date_lt1 = shift_fn(backlog.plan_date, 1) if shift_fn else backlog.plan_date
750 date_lt2 = shift_fn(backlog.plan_date, 2) if shift_fn else backlog.plan_date
751
752 checked_dates = set()
753 for parent_date in [date_lt2, date_lt1]:
754 if parent_date in checked_dates:
755 continue
756 checked_dates.add(parent_date)
757
758 plan_rows = list(LineBacklog.objects.filter(
759 product=parent_product,
760 plan_date=parent_date,
761 sequence_no__gt=0,
762 ).values_list('sequence_no', 'plan_qty'))
763 if not plan_rows:
764 continue
765
766 def _sort_key(item):
767 seq = int(item[0] or 0)
768 return seq if seq > 0 else 10 ** 9
769 plan_rows.sort(key=_sort_key)
770
771 for lot_index, (seq_no, plan_qty) in enumerate(plan_rows):
772 seq = int(seq_no or 0)
773 force_pm = seq > FLOOR_SHIPPING_PM_THRESHOLD
774 effective_lt = 1 if force_pm else (2 if lot_index == 0 else 1)
775 expected_parent = shift_fn(backlog.plan_date, effective_lt) if shift_fn else backlog.plan_date
776 if expected_parent != parent_date:
777 continue
778 plan = int(plan_qty or 0)
779 if plan:
780 total += Decimal(str(plan)) * qty_per
781
782 return total

844def _compute_planned_stock_lt_adjustment(
845 product_id,
846 initial_date,
847 max_lt,
848 shift_fn,
849 demand_map=None,
850 final_delivery_lt=None,
851 line_id=None,
852):
853 """
854 計画在庫初期値のLT調整量を計算する。
855
856 実在庫はLTシフトなしで計算されるが、計画在庫はLTシフトあり。
857 そのため、initial_dateの実在庫を計画在庫の初期値として使う際は、
858 「initial_date+1 ～ initial_date+LT」の出庫分（LTシフト）を差し引く必要がある。
859
860 社内品: 親製品の実績（BOM LT 分）を差し引く
861 最終品/ライン最終品: デリバリLT分の需要を差し引く
862
863 ※ initial_date+LT ≤ today-1 の条件下では全て実績確定しているため値が安定する。
864 """
865 from masters.models import BOMItem
866 from django.db.models import Sum
867
868 total_adjustment = Decimal('0')
869
870 # 最終品/ライン最終品：calc_start_date（initial_date）から後ろ向きに
871 # デリバリLT日分の需要を差し引く。
872 #
873 # 最終品は demand_map（LineDemand由来）を優先し、
874 # ライン最終品など demand_map を持たないケースでは同日の order_qty を使用する。
875 # LT 分だけ在庫の初期値から差し引いて計画在庫の初期値とする：
876 # LT=1: demand[initial_date]
877 # LT=2: demand[initial_date] + demand[initial_date - 1営業日]
878 # LT=N: demand[initial_date] + ... + demand[initial_date - (N-1)営業日]
879 #
880 # これにより初期値の段階で LT 分だけ計画在庫 < 実在庫 となり、
881 # 以降は普通の計算式（LT シフトなし）で継続する。
882 if final_delivery_lt is not None:
883 delivery_lt = int(final_delivery_lt or 0)
884 else:
885 # 呼び出し元未指定時は0日として扱う（後方互換のself_lt_days参照は廃止）
886 delivery_lt = 0
887
888 if delivery_lt >= 0:
889 for offset in range(delivery_lt): # 0, 1, ..., delivery_lt-1
890 adj_date = shift_fn(initial_date, -offset) # initial_date から後ろ向きにシフト
891 qty = demand_map.get((product_id, adj_date), Decimal('0')) if demand_map else Decimal('0')
892 if not qty and line_id:
893 backlog_rows = LineBacklog.objects.filter(
894 line_id=line_id,
895 product_id=product_id,
896 plan_date=adj_date,
897 )
898 qty = Decimal(str(sum(int(row.order_qty or 0) for row in backlog_rows)))
899 total_adjustment += Decimal(str(qty))
900 return int(total_adjustment)
901
902 # 社内品：親の BOM 経由でLTシフト分の実績出庫を差し引く
903 parent_bom_items = BOMItem.objects.filter(
904 child_product_id=product_id,
905 bom__is_active=True,
906 bom__is_coproduct=False,
907 ).select_related('bom__parent_product')
908
909 if not parent_bom_items.exists():
910 return 0
911
912 total_adjustment = Decimal('0')
913 for bom_item in parent_bom_items:
914 parent_product = bom_item.bom.parent_product
915 if not parent_product:
916 continue
917 qty_per = bom_item.quantity or Decimal('0')
918 if qty_per == 0:
919 continue
920 lead_days = bom_item.lead_time_days or 0
921 if lead_days == 0:
922 continue
923
924 for offset in range(1, lead_days + 1):
925 adj_date = shift_fn(initial_date, offset)
926 downstream_groups = LineBacklog.objects.filter(
927 product=parent_product,
928 plan_date=adj_date,
929 ).values('product_id', 'line_id', 'process_id', 'plan_date').annotate(
930 actual_total=Sum('actual_qty'),
931 )
932 for downstream in downstream_groups:
933 actual = int(downstream.get('actual_total') or 0)
934 scrap = _get_shipment_scrap_qty(
935 downstream['product_id'],
936 downstream['line_id'],
937 downstream['process_id'],
938 downstream['plan_date'],
939 )
940 use_qty = Decimal(str(actual)) + scrap
941 if use_qty:
942 total_adjustment += use_qty * Decimal(str(qty_per))
943
944 return int(total_adjustment)

1707 is_line_final = bool(getattr(sample.product, 'is_line_final_product', False))

1713 )

1718 planned_shipment = Decimal(str(order_total))

1724 else:
1725 if forced_parent_shipment_mode == 'ORDER_QTY':
1726 planned_shipment = Decimal(str(order_total))

1728 planned_shipment = _calculate_parent_actual_or_plan_shipment(sample, shift_working_days, floor_shipping_cache)
1729 else:
1730 planned_shipment = _calculate_parent_planned_shipment(sample, business_today, shift_working_days, floor_shipping_cache)

1732
1733 # 在庫も進度と同様、休日を含めて「前日（暦日）」を基準に引き継ぐ。
1734 prev_day = plan_date - timedelta(days=1)
1735 prev_planned = planned_by_date.get(prev_day, last_planned)
1736 planned_stock_adjust = _resolve_day_adjustment(rows, planned_stock_adjust_map)
1737
1738 # 計画在庫 = 前日計画在庫 + 実績/計画 - 出庫
1739 # 仕損による子部品消費は出庫計算で反映済み（adjust_qty は使用しない）
1740 if plan_date < business_today:
1741 planned_stock = (
1742 prev_planned
1743 + actual_total
1744 - planned_shipment
1745 + planned_stock_adjust
1746 )
1747 else:
1748 planned_stock = (
1749 prev_planned
1750 + plan_total
1751 - planned_shipment
1752 + planned_stock_adjust
1753 )
1754
1755 rep = pick_representative(rows)
1756 for row in rows:
1757 row.planned_stock_qty = 0
1758 rep.planned_stock_qty = planned_stock
1759 planned_by_date[plan_date] = planned_stock
1760 last_planned = planned_stock
1761
1762 # 更新対象に追加
1763 backlogs_to_update.extend(rows)
1764
1765 if plan_date < business_today:
1766 trace_log(
1767 line_id, product_id, plan_date,
1768 f'計画在庫計算(過去日/実績ベース): 前日={prev_planned}, 当日実績={actual_total}, '
1769 f'当日出庫={planned_shipment}, 手動調整={planned_stock_adjust}, 結果={planned_stock}'
1770 )
1771 else:
1772 trace_log(
1773 line_id, product_id, plan_date,
1774 f'計画在庫計算(当日以降/計画ベース): 前日={prev_planned}, 当日計画={plan_total}, '
1775 f'当日出庫={planned_shipment}, 手動調整={planned_stock_adjust}, 結果={planned_stock}'
1776 )
1777
1778 if backlogs_to_update:
1779 LineBacklog.objects.bulk_update(backlogs_to_update, ['planned_stock_qty'])
